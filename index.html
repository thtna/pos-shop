📜 Hướng dẫn cầu nối in cho máy in LAN (chạy 1 lần trên máy thu ngân) 
1. Cài Node.js trên máy thu ngân
2. Lưu file print-bridge.js (nội dung bên dưới)
3. Chạy:  node print-bridge.js
4. Quay lại đây, chọn kiểu "LAN", nhập: http://localhost:9100/print

----- print-bridge.js -----
const http = require('http');
const net = require('net');
http.createServer((req, res) => {
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Access-Control-Allow-Headers', '*');
  if (req.method === 'OPTIONS') return res.end();
  let body = [];
  req.on('data', c => body.push(c));
  req.on('end', () => {
    const data = Buffer.concat(body);
    const host = req.headers['x-printer-ip'] || '192.168.1.100';
    const sock = net.connect(9100, host, () => sock.end(data));
    sock.on('error', e => { console.error(e.message); res.statusCode = 500; res.end('ERR'); });
    sock.on('close', () => res.end('OK'));
    console.log('Đã in tới', host, data.length, 'bytes');
  });
}).listen(9100, () => console.log('Cầu nối in đang chạy tại cổng 9100'));
